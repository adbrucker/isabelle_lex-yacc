%{
#include <stdio.h>
void yyerror(const char *s);
int yylex(void);
%}

/* Token-Definitionen für Rust-Strukturen */
%token FN LET MUT IF ELSE MATCH RETURN PANIC
%token IDENT NUMBER LITERAL
%token ARROW "->"
%token ANDMUT "&mut"
%token FATARROW "=>"

/* Operatorpräzedenzen */
%left '||'
%left '&&'
%left '==' '!=' '<' '>'
%left '+' '-'
%left '*' '/'

%%

/* Funktionen und Top-Level Strukturen */
item:
      fn_decl
    ;

fn_decl:
      FN IDENT '(' param_list ')' opt_ret_type block
    ;

param_list:
      /* leer */
    | params
    ;


params:
      param
    | params ',' param
    ;

param:
      IDENT ':' type_expr
    ;

opt_ret_type:
      /* leer (unit) */
    | ARROW type_expr
    ;

type_expr:
      IDENT
    | ANDMUT IDENT
    ;

/* Funktionsrumpf und Anweisungen */
block:
      '{' stmt_list '}'
    ;

stmt_list:
      /* leer */
    | stmt_list stmt
    ;

stmt:
      LET opt_mut IDENT ':' type_expr '=' expr ';'
    | IDENT '=' expr ';'
    | expr ';'
    | block
    ;


opt_mut:
      /* leer */
    | MUT
    ;

/* Ausdrücke (Expressions) */
expr:
      IDENT
    | NUMBER
    | LITERAL
    | expr '+' expr
    | expr '-' expr
    | expr '&&' expr

    | expr '||' expr
    | expr '==' expr
    | PANIC '!' '(' LITERAL ')'          /* Kritisch für Verifikation (Panic-Pfad) */
    | IF expr block ELSE block
    | MATCH expr '{' match_branches '}'
    ;

match_branches:
      match_branch
    | match_branches match_branch
    ;

match_branch:
      pattern FATARROW expr ','
    ;

pattern:
      IDENT
    | NUMBER
    | '_'
    ;



%%

void yyerror(const char *s) {
    fprintf(stderr, "Syntaxfehler: %s\n", s);
}
